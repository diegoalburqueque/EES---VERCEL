var TOKEN=sessionStorage.getItem("atm_token")||"";
var YO=null, FICHA=null;

// --- utilidades ---
function llamar(fn,args){
  return new Promise(function(ok,err){
    google.script.run.withSuccessHandler(ok).withFailureHandler(function(e){err(e.message||String(e));})[fn].apply(null,args||[]);
  });
}
// Con varios calificadores trabajando a la vez, dos pueden guardar en el mismo
// instante y el segundo recibe "el sistema está guardando otra resolución".
// Eso no es un error del usuario: es una espera. Se reintenta solo, en silencio,
// y sólo se muestra el mensaje si después de tres intentos sigue ocupado.
function llamarConEspera(fn,args,intentos){
  var restantes=(intentos===undefined?3:intentos);
  return llamar(fn,args).catch(function(e){
    var ocupado=String(e).indexOf("guardando otra resoluci")>=0;
    if(!ocupado||restantes<=1)throw e;
    return new Promise(function(ok){setTimeout(ok,2500);})
      .then(function(){return llamarConEspera(fn,args,restantes-1);});
  });
}
function esc(v){return String(v==null?"":v).replace(/&/g,"&amp;").replace(/</g,"&lt;").replace(/>/g,"&gt;").replace(/"/g,"&quot;");}
function msg(texto,tipo){
  document.getElementById("msg").innerHTML=texto?'<div class="msg '+(tipo||"error")+'">'+esc(texto)+'</div>':"";
  if(texto)window.scrollTo(0,0);
}
function ver(id){
  ["vLogin","vBandeja","vFicha","vAdmin","vRetro","vKpi","vCalidad"].forEach(function(v){document.getElementById(v).classList.add("oculto");});
  document.getElementById(id).classList.remove("oculto");
}
// Qué botones ve cada rol, en un solo lugar. La lista estaba escrita dos veces
// —al entrar y al restaurar la sesión— y el botón nuevo aparecía sólo si habías
// tecleado la contraseña en esa pestaña.
var BOTONES_ROL={ADMIN:["btnAdmin","btnRetro","btnKpi","btnCalidad"],CALIDAD:["btnCalidad"]};
function botonesPorRol(rol){
  (BOTONES_ROL[String(rol||"").toUpperCase()]||[]).forEach(function(b){
    document.getElementById(b).classList.remove("oculto");});
}
function chipDictamen(d){
  var t=String(d||"").toUpperCase();
  var c=t.indexOf("MANTIENE")>=0?"mant":t.indexOf("AUMENTA")>=0?"aum":t.indexOf("DISMINUYE")>=0?"dism":"pend";
  return '<span class="chip '+c+'">'+esc(d||"sin dictamen")+'</span>';
}

// --- sesión ---
function entrar(){
  msg("");
  var u=document.getElementById("lUsuario").value,c=document.getElementById("lClave").value;
  if(!u||!c){msg("Ingresa usuario y contraseña.");return;}
  llamar("apiLogin",[u,c]).then(function(r){
    TOKEN=r.token; YO=r; sessionStorage.setItem("atm_token",TOKEN);
    document.getElementById("cab").classList.remove("oculto");
    document.getElementById("quien").textContent=r.nombre+" · "+r.rol;
    botonesPorRol(r.rol);
    irBandeja();
  }).catch(function(e){msg(e);});
}
function salir(){
  llamar("apiCerrarSesion",[TOKEN]).catch(function(){});
  TOKEN="";YO=null;sessionStorage.removeItem("atm_token");
  document.getElementById("cab").classList.add("oculto");
  ver("vLogin");msg("");
}

// --- bandeja ---
function irBandeja(){
  msg("");ver("vBandeja");
  document.getElementById("tPendientes").innerHTML='<div class="cargando">Cargando…</div>';
  document.getElementById("tResueltos").innerHTML="";
  llamar("apiBandeja",[TOKEN]).then(function(d){
    YO=d.usuario;
    document.getElementById("tPendientes").innerHTML=tablaBandeja(d.pendientes,false);
    document.getElementById("tResueltos").innerHTML=tablaBandeja(d.resueltos,true);
  }).catch(function(e){msg(e);});
}
function tablaBandeja(lista,resuelto){
  if(!lista||!lista.length)return '<p class="mini">Sin trámites en esta bandeja.</p>';
  var h='<table><tr><th style="width:90px">Trámite</th><th>Usuario evaluado</th><th style="width:150px">Dictamen ATM</th>'+
        '<th style="width:80px">IVADEC</th><th style="width:80px">Propuesta</th>'+
        (resuelto?'<th style="width:90px">Resolución</th>':'')+'<th style="width:100px"></th></tr>';
  lista.forEach(function(t){
    RESUMEN_BANDEJA[t.id]=t;   // para pintar la ficha al instante al abrirla
    h+='<tr><td><b>'+esc(t.id)+'</b></td><td>'+esc(t.nombre)+'<br><span class="mini">'+esc(t.run)+'</span></td>'+
       '<td>'+chipDictamen(t.dictamen)+'</td><td>'+esc(t.pctIvadec)+'%</td>'+
       '<td><b>'+esc(t.pctProp)+'%</b><br><span class="mini">'+esc(t.gradoProp)+'</span></td>'+
       (resuelto?'<td>'+(t.estado==="MODIFICADO"
           ? '<span class="chip obs">MODIFICADO</span><br><span class="mini">'+esc(t.motivo)+'</span>'
           : t.estado==="NO EVALUABLE"
             ? '<span class="chip dism">NO EVALUABLE</span>'
             : t.estado==="OBSERVADO"
               ? '<span class="chip dism">OBSERVADO</span><br><span class="mini">'+esc(t.motivo)+'</span>'
               : t.estado==="CERRADO"
                 ? '<span class="chip mant">CERRADO</span><br><span class="mini">final '+esc(t.pctFinal)+'%</span>'
                 : '<span class="chip mant">RATIFICADO</span>')+'</td>':'')+
       '<td><button onclick="abrirFicha(\''+esc(t.id)+'\')">'+(resuelto?"Ver":"Resolver")+'</button></td></tr>';
  });
  return h+'</table>';
}

// --- ficha ---
// La bandeja ya conoce nombre, dictamen y porcentajes de cada trámite. Pintarlos
// de inmediato mientras el servidor trae el resto hace que la ficha se sienta
// instantánea, aunque el expediente completo tarde un segundo más en llegar.
var RESUMEN_BANDEJA={};
function abrirFicha(id){
  msg("");ver("vFicha");
  var t=RESUMEN_BANDEJA[id]||{};
  document.getElementById("vFicha").innerHTML=
    '<div class="panel"><h2>Trámite '+esc(id)+(t.nombre?' · '+esc(t.nombre):'')+'</h2>'+
    (t.run?'<p class="mini">RUN '+esc(t.run)+'</p>':'')+
    (t.dictamen?'<p>'+chipDictamen(t.dictamen)+'</p>':'')+
    (t.pctProp?'<table><tr><th style="width:28%">&nbsp;</th><th>Porcentaje</th><th>Grado</th></tr>'+
      '<tr><th>Según IVADEC-CIF</th><td>'+esc(t.pctIvadec)+'%</td><td>—</td></tr>'+
      '<tr><th>Propuesta ATM</th><td><b>'+esc(t.pctProp)+'%</b></td><td><b>'+esc(t.gradoProp)+'</b></td></tr></table>':'')+
    '</div><div class="panel cargando">Cargando el expediente completo…</div>';
  llamar("apiFicha",[TOKEN,id]).then(function(f){FICHA=f;pintarFicha(f);}).catch(function(e){msg(e);ver("vBandeja");});
}
function pintarFicha(f){
  var p=f.propuesta,resuelto=!!(f.resolucion.decision||"").trim();
  var D=f.documentos||{grupos:[],atm:[],avisos:[],faltantes:[]};
  var h='';

  h+='<div class="panel"><h2>Trámite '+esc(f.id)+' · '+esc(f.identificacion.nombre)+'</h2>'+
     '<table><tr><th style="width:22%">RUN</th><td>'+esc(f.identificacion.run)+'</td>'+
     '<th style="width:22%">Procesado por el ATM</th><td>'+esc(f.identificacion.fechaProceso)+'</td></tr>'+
     '<tr><th>Origen</th><td>'+esc(f.identificacion.origen)+'</td>'+
     '<th>Movilidad reducida</th><td>'+esc(f.identificacion.movilidad)+'</td></tr></table></div>';

  h+='<div class="panel"><h2>Documentos del expediente</h2>';
  if(D.faltantes&&D.faltantes.length)
    h+='<div class="msg error">Falta en el expediente: '+esc(D.faltantes.join(", "))+'. Verifícalo antes de resolver.</div>';
  (D.avisos||[]).forEach(function(a){ h+='<div class="msg aviso">'+esc(a)+'</div>'; });
  (D.grupos||[]).forEach(function(g){
    h+='<h3>'+esc(g.etiqueta)+(g.cantidad?' <span class="mini">('+g.cantidad+')</span>':'')+
       (g.duplicado?' <span class="chip dism">duplicado</span>':'')+'</h3><div class="docs">';
    if(g.documentos.length)g.documentos.forEach(function(d){
      h+='<a href="'+esc(d.url)+'" target="_blank">'+esc(d.nombre)+'</a>';
    });
    else h+='<span class="chip '+(g.obligatorio?"dism":"pend")+'">'+(g.obligatorio?"NO ESTÁ EN EL EXPEDIENTE":"sin documentos complementarios")+'</span>';
    h+='</div>';
  });
  h+='<h3>Documentos emitidos por el ATM</h3><div class="docs">';
  (D.atm||[]).forEach(function(d){ if(d.url)h+='<a href="'+esc(d.url)+'" target="_blank">'+esc(d.tipo)+'</a>'; });
  h+='</div></div>';

  // Todo el expediente narrado: el calificador valida sin abrir documentos.
  var A=f.antecedentes||{};
  if(A.error)h+='<div class="msg aviso">No se pudo leer el detalle ampliado del expediente ('+esc(A.error)+'). Revisa los documentos directamente.</div>';
  else{
    var fil=function(items){
      return '<table>'+items.map(function(i){
        return '<tr><th style="width:26%">'+esc(i.rotulo)+'</th><td>'+esc(i.valor).replace(/\n/g,"<br>")+'</td></tr>';
      }).join("")+'</table>';
    };
    h+='<div class="panel"><h2>Antecedentes para validar</h2>'+
       '<p class="mini">Todo lo que sigue fue extraído del expediente por el ATM. No necesitas abrir los documentos para validar.</p>'+
       '<h3>Antecedentes sociales relevantes</h3>'+fil(A.sociales||[])+
       '<h3>Datos relevantes de calificación · IBF</h3>'+fil(A.biomedicos||[])+
       '<h3>Informes o exámenes complementarios</h3>';
    // Mismo criterio que el generador de la propuesta: si el motor ya leyó los
    // complementarios se muestran sus hallazgos; el texto fijo ("NO" o el
    // resumen de concordancia) sólo aparece cuando no hay nada que detallar.
    var C=A.complementarios||{items:[]};
    if(C.texto)h+='<p>'+esc(C.texto)+'</p>';
    else if((C.items||[]).length){
      h+='<ul>'+C.items.map(function(d){return '<li>'+esc(d.linea||d.nombre)+'</li>';}).join("")+'</ul>';
      if(C.concordantes)h+='<p class="mini">El resto de los informes son concordantes con diagnóstico de IBF.</p>';
      if(C.pendiente)h+='<div class="msg aviso">Este expediente se procesó con una versión anterior del ATM, que no leía el contenido de los complementarios. Revísalos tú y confirma si aportan un diagnóstico no informado en el IBF. Para que el ATM los analice, hay que reprocesar el trámite.</div>';
    }else h+='<p>NO</p>';
    h+='<h3>IVADEC</h3>'+fil(A.ivadec||[])+
       '<h3>Observaciones del IVADEC</h3><ul>'+(A.observacionesIvadec||[]).map(function(o){return '<li>'+esc(o)+'</li>';}).join("")+'</ul>'+
       '<h3>Movilidad reducida · algoritmo oficial</h3>'+
       '<table><tr><th style="width:26%">Resultado</th><td><b>'+esc((A.movilidad||{}).resultado)+'</b>'+((A.movilidad||{}).provisional?' (PROVISIONAL)':'')+'</td></tr>'+
       '<tr><th>Ruta</th><td>'+esc((A.movilidad||{}).ruta||"no resuelta")+'</td></tr>'+
       '<tr><th>Lecturas</th><td>'+esc(((A.movilidad||{}).respuestas||[]).join(" | ")||"sin lecturas")+'</td></tr></table>';
    if((A.reglas||[]).length){
      h+='<h3>Por qué el ATM propone esto</h3><table><tr><th style="width:80px">Regla</th><th style="width:70px">Efecto</th><th>Hallazgo</th></tr>'+
         A.reglas.map(function(r){return '<tr><td><b>'+esc(r.codigo)+'</b></td><td>'+esc(r.efecto)+'</td><td>'+esc(r.texto)+'</td></tr>';}).join("")+'</table>';
    }
    if(A.fundamentoAtm)h+='<h3>Fundamento del dictamen</h3><p>'+esc(A.fundamentoAtm)+'</p>';
    h+='</div>';
  }

  h+='<div class="panel"><h2>Propuesta del ATM</h2>'+
     '<p>'+chipDictamen(p.dictamen)+' &nbsp; <span class="mini">'+esc(p.perfil)+'</span></p>'+
     '<table><tr><th style="width:28%"></th><th>IDIS</th><th>Porcentaje</th><th>Grado</th></tr>'+
     '<tr><th>Según IVADEC-CIF</th><td>'+esc(p.idisIvadec)+'</td><td>'+esc(p.pctIvadec)+'%</td><td>'+esc(p.gradoIvadec)+'</td></tr>'+
     '<tr><th>Propuesta ATM</th><td><b>'+esc(p.idisProp)+'</b></td><td><b>'+esc(p.pctProp)+'%</b></td><td><b>'+esc(p.gradoProp)+'</b></td></tr></table>'+
     '<h3>Reglas de ajuste aplicadas</h3><p>'+esc(p.reglas||"ninguna")+'</p>'+
     '<h3>Fundamento</h3><p>'+esc(p.fundamento)+'</p></div>';

  if(f.alertas.length||f.cruces.length||f.observaciones.length){
    h+='<div class="panel"><h2>Puntos a revisar</h2>';
    if(f.alertas.length){h+='<h3>Alertas</h3><ul>';f.alertas.forEach(function(a){h+='<li>'+esc(a)+'</li>';});h+='</ul>';}
    if(f.cruces.length){
      h+='<h3>Cruces discordantes</h3><table><tr><th style="width:70px">Cruce</th><th>Campo A</th><th>Campo B</th><th>Fundamento</th></tr>';
      f.cruces.forEach(function(c){h+='<tr><td>'+esc(c.cruce)+'</td><td>'+esc(c.campoA)+'</td><td>'+esc(c.campoB)+'</td><td>'+esc(c.fundamento)+'</td></tr>';});
      h+='</table>';
    }
    if(f.observaciones.length){h+='<h3>Observaciones del proceso</h3><ul>';f.observaciones.forEach(function(o){h+='<li>'+esc(o)+'</li>';});h+='</ul>';}
    h+='</div>';
  }

  if(resuelto){
    var rt=f.retro;
    h+='<div class="panel"><h2>Resolución registrada</h2>'+
       '<table><tr><th style="width:22%">Decisión</th><td>'+esc(f.resolucion.decision)+'</td>'+
       '<th style="width:22%">Resultado</th><td>IDIS '+esc(f.resolucion.idis)+' · '+esc(f.resolucion.pct)+'% · '+esc(f.resolucion.grado)+'</td></tr>'+
       (rt?'<tr><th>Motivo del rechazo</th><td colspan="3"><b>'+esc(rt.motivo)+'</b> · '+esc(rt.motivoTexto)+'</td></tr>':'')+
       '<tr><th>Fundamento</th><td colspan="3">'+esc(f.resolucion.fundamento||"—")+'</td></tr>'+
       '<tr><th>Calificador</th><td>'+esc(f.resolucion.calificador)+'</td><th>Fecha</th><td>'+esc(f.resolucion.fecha)+'</td></tr></table>';
    if(rt){
      h+='<h3>Seguimiento de Salud 360</h3>'+
         (rt.estadoContacto==="CERRADO"
           ? '<table><tr><th style="width:22%">Contacto</th><td>'+esc(rt.contactadoPor)+' · '+esc(rt.fechaContacto)+'</td></tr>'+
             '<tr><th>Resultado</th><td>'+esc(rt.resultado)+'</td></tr>'+
             '<tr><th>Propuesta final a COMPIN</th><td><b>IDIS '+esc(rt.idisFinal)+' · '+esc(rt.pctFinal)+'% · '+esc(rt.gradoFinal)+'</b></td></tr>'+
             (rt.fundamentoFinal?'<tr><th>Fundamento final</th><td>'+esc(rt.fundamentoFinal)+'</td></tr>':'')+'</table>'
           : '<div class="msg aviso">Tu observación quedó registrada. Salud 360 te va a contactar para revisarla; de esa conversación sale la propuesta definitiva que va a COMPIN.</div>');
    }
    h+='<p><button onclick="irBandeja()">Volver</button></p></div>';
  }else{
    h+='<div class="panel"><h2>Tu resolución</h2>'+
       '<div class="fila"><div><label>Movilidad reducida (MR) que va en la propuesta</label>'+
       '<select id="mMR"><option value="SÍ"'+(/^S/i.test(p.movilidad||f.identificacion.movilidad||"")?" selected":"")+'>SÍ</option>'+
       '<option value="NO"'+(/^S/i.test(p.movilidad||f.identificacion.movilidad||"")?"":" selected")+'>NO</option></select></div>'+
       '<div><label>Reevaluación (REEV)</label><select id="mREEV">'+

function resolver(decision){
  msg("");
  var pct=null,fund="",motivo="";
  if(decision==="NO_EVALUABLE"){
    motivo=document.getElementById("neCausa").value;
    fund=document.getElementById("neDetalle").value;
    if(!motivo){msg("Selecciona la causa por la que el caso no se puede evaluar.");return;}
    if(fund.trim().length<20){msg("Describe qué impide evaluarlo (mínimo 20 caracteres). El administrador lo necesita para resolverlo.");return;}
    if(!confirm("Vas a declarar el trámite "+FICHA.id+" como NO EVALUABLE.\n\nNo se emite documento y el caso pasa al administrador. ¿Confirmas?"))return;
  }else if(decision==="MODIFICA"){
    motivo=document.getElementById("mMotivo").value;
    pct=document.getElementById("mPct").value;
    fund=document.getElementById("mFund").value;
    if(!motivo){msg("Selecciona el motivo por el que no aceptas la propuesta.");return;}
    if(!pct){msg("Selecciona el porcentaje que corresponde.");return;}
    if(fund.trim().length<20){msg("Explica el motivo por escrito (mínimo 20 caracteres).");return;}
  }else{
    if(!confirm("Vas a RATIFICAR la propuesta del ATM tal cual y cerrar el trámite "+FICHA.id+".\n\nEl documento va a quedar firmado con tu nombre y profesión. ¿Confirmas?"))return;
  }
  var mr=(document.getElementById("mMR")||{}).value||"",
      reev=(document.getElementById("mREEV")||{}).value||"NO";
  document.querySelectorAll("#vFicha button").forEach(function(b){b.disabled=true;});
  msg("Guardando…","aviso");
  llamarConEspera("apiResolver",[TOKEN,FICHA.id,decision,pct,fund,motivo,mr,reev]).then(function(r){
    if(r.estado==="NO EVALUABLE"){
      msg("Trámite "+FICHA.id+" declarado NO EVALUABLE ("+r.causa+": "+r.causaTexto+"). Pasó al administrador.","aviso");
      irBandeja();
    }else{
      // Ratificar y modificar terminan igual: documento emitido y a Cero Fila.
      pantallaCeroFilas(r);
    }
  }).catch(function(e){
    msg(e);
    document.querySelectorAll("#vFicha button").forEach(function(b){b.disabled=false;});
  });
}

function resolverNE(id,decision){
  var det=document.getElementById("ne_"+id).value;
  if(det.trim().length<20){msg("Explica qué resolviste (mínimo 20 caracteres): queda en el historial del caso.");return;}
  var aviso=decision==="DEVOLVER"
    ? "El trámite "+id+" vuelve a la bandeja del mismo calificador con la propuesta del ATM intacta."
    : "El trámite "+id+" se cierra SIN calificar. No se emitirá documento ni se enviará a Cero Fila.";
  if(!confirm(aviso+"\n\n¿Confirmas?"))return;
  msg("Guardando…","aviso");
  llamarConEspera("apiAdminResolverNoEvaluable",[TOKEN,id,decision,det]).then(function(){
    msg("Trámite "+id+" "+(decision==="DEVOLVER"?"devuelto a la cola.":"cerrado sin calificar."),"bien");
    irRetro();
  }).catch(function(e){msg(e);});
}
function cerrarRetro(id){
  var res=document.getElementById("rr_"+id).value,
      pct=document.getElementById("rp_"+id).value,
      fund=document.getElementById("rf_"+id).value;
  if(res.trim().length<20){msg("Registra el resultado del contacto (mínimo 20 caracteres).");return;}
  if(!pct){msg("Selecciona el porcentaje final que va a COMPIN.");return;}
  if(!confirm("Vas a fijar la propuesta FINAL del trámite "+id+" en "+pct+"%. Es la que va a COMPIN. ¿Confirmas?"))return;
  msg("Guardando…","aviso");
  llamarConEspera("apiAdminCerrarRetro",[TOKEN,id,res,pct,fund]).then(function(r){
    msg("Trámite "+id+" cerrado. Propuesta final a COMPIN: IDIS "+r.idis+" · "+r.pct+"% · "+r.grado+".","bien");
    irRetro();
  }).catch(function(e){msg(e);});
}

// Una barra de proporción vale más que el número solo: con doce calificadores
// en una tabla, la vista compara sola.
function barraKpi(pct,suficiente){
  var color=suficiente?(pct>=80?"var(--ok)":pct>=60?"var(--obs)":"var(--mal)"):"#cbd5e1";
  return '<div style="background:#e2e8f0;border-radius:6px;height:10px;width:110px;display:inline-block;vertical-align:middle">'+
         '<div style="background:'+color+';width:'+Math.max(0,Math.min(100,pct))+'%;height:10px;border-radius:6px"></div></div>';
}
function filasKpi(lista,rotulo,minimo){
  if(!lista.length)return '<p class="mini">Sin datos todavía.</p>';
  var h='<table><tr><th>'+esc(rotulo)+'</th><th style="width:150px">Tasa de aceptación</th>'+
        '<th style="width:70px">Acepta</th><th style="width:70px">Rechaza</th>'+
        '<th style="width:80px">Decisiones</th><th style="width:80px">No evaluables</th></tr>';
  lista.forEach(function(c){
    h+='<tr><td>'+esc(c.clave)+'</td>'+
       '<td>'+barraKpi(c.tasa,c.suficiente)+' <b>'+c.tasa+'%</b>'+
       (c.suficiente?'':' <span class="chip pend" title="Menos de '+minimo+' decisiones">pocas</span>')+'</td>'+
       '<td>'+c.aceptados+'</td><td>'+c.rechazados+'</td><td>'+c.decisiones+'</td><td>'+c.noEvaluables+'</td></tr>';
  });
  return h+'</table>';
}

// El reloj del contrato por caso: 20 días hábiles desde que se asignó. Rojo
// cuando ya se pasó, naranjo cuando quedan 3 días o menos.
function chipPlazo(dias){
  if(dias===null||dias===undefined)return '<span class="chip pend">sin fecha de asignación</span>';
  if(dias<0)return '<span class="chip dism">vencido hace '+Math.abs(dias)+' d. hábiles</span>';
  if(dias<=3)return '<span class="chip aum">quedan '+dias+' d. hábiles</span>';
  return '<span class="chip mant">quedan '+dias+' d. hábiles</span>';
}

function revisarCC(id,decision){
  var obs=document.getElementById("cc_"+id).value,
      causal=document.getElementById("cs_"+id).value;
  if(decision==="DEVUELVE"){
    if(!causal){msg("Selecciona la causal del Art. 40° por la que devuelves el caso.");return;}
    if(obs.trim().length<20){msg("Explica qué hay que corregir (mínimo 20 caracteres).");return;}
    if(!confirm("El trámite "+id+" vuelve al calificador y su resolución se reabre para que pueda corregirla.\n\n¿Confirmas?"))return;
  }
  msg("Guardando…","aviso");
  llamarConEspera("apiCalidadRevisar",[TOKEN,id,decision,causal,obs]).then(function(r){
    msg(r.mensaje,"bien");irCalidad();
  }).catch(function(e){msg(e);});
}
